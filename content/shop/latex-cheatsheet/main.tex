\sheetsection{Document skeleton}
\begin{lstlisting}
\documentclass[11pt, a4paper]{article}
\usepackage[english]{babel}
\usepackage{graphicx, amsmath}

\title{My Report}
\author{First Last}
\date{\today}

\begin{document}
\maketitle
\tableofcontents

\section{Introduction}
Text goes here.  % a comment

\end{document}
\end{lstlisting}

\verb|beamer|  & presentation slides \\
\end{cmds}

\sheetsection{Lists}
\begin{lstlisting}

\end{lstlisting}
\begin{cmds}[0.4]

\verb|scale=0.5|, \verb|angle=90| & scale, rotate \\

\end{cmds}

\verb|\cmidrule{2-3}|      & partial rule (booktabs) \\
\end{cmds}
{\color{subtle}Captions go \emph{above} tables and \emph{below} figures.}

\sheetsection{References (BibLaTeX, APA)}
\begin{lstlisting}
\usepackage[style=apa, backend=biber]{biblatex}
\addbibresource{references.bib}
...
As \textcite{knuth1984} shows \ldots
\ldots{} in print \parencite[p.~12]{knuth1984}.
\printbibliography
\end{lstlisting}
\begin{cmds}[0.42]
\verb|\textcite{key}|          & Knuth (1984) \\
\verb|\parencite{key}|         & (Knuth, 1984) \\
\verb|\parencite[p.~12]{key}|  & (Knuth, 1984, p.\,12) \\
\end{cmds}
\sheetsub{An entry in references.bib}
\begin{lstlisting}[language={}]
@book{knuth1984,
  author    = {Knuth, Donald E.},
  title     = {The TeXbook},
  publisher = {Addison-Wesley},
  year      = {1984},
}
\end{lstlisting}
{\color{subtle}Other types: \texttt{@article}, \texttt{@online} (with \texttt{url}, \texttt{urldate}), \texttt{@inproceedings}, \texttt{@thesis}. Google Scholar's ``Cite'' $\to$ BibTeX gives you an entry to paste.}

\sheetsection{Errors and what they mean}
\begin{cmds}[0.4]
\textbf{Undefined control sequence} & typo, or the package is not loaded \\
\textbf{Missing \$ inserted}        & \verb|_| or \verb|^| outside math: use \verb|\_| or \verb|$x^2$| \\
\textbf{File \ldots{} not found}    & wrong file name or folder \\
\textbf{Citation \ldots{} undefined} & run Biber, or the key is misspelt \\
\textbf{Misplaced alignment tab \&} & write \verb|\&| outside tables \\
\textbf{Overfull \textbackslash hbox} & a line sticks out: rephrase it \\
\textbf{Runaway argument}           & a \verb|}| is missing \\
\end{cmds}
{\color{subtle}Read the \emph{first} error in the log: the rest usually follow from it.}
